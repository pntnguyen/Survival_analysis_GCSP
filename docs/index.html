Pathway scoring

Standardize expression of each genes and calculate pathway score as below:

\[ S_{i,p} = \frac{1}{|G_p|} \sum_{g \in G_p} Z_{ig} \]

where:

  • \(i\): patient
  • \(p\): pathway
  • \(G_p\): genes belonging to pathway \(p\) and present in METABRIC
  • \(Z_{ig}\): standardized expression of gene \(g\) in patient \(i\)
Code 
# 1. Map each available gene to its actual column name in the data
col_lookup <- tibble(col = names(df), gene_lower = tolower(names(df)))

pathway_cols <- pathway_genes |>
  filter(in_data) |>
  inner_join(col_lookup, by = "gene_lower")   # adds the `col` column

# 2. Z-score each gene column (mean 0, SD 1 across patients)
gene_cols <- unique(pathway_cols$col)

z_mat <- df |>
  select(all_of(gene_cols)) |>
  mutate(across(everything(), ~ as.numeric(scale(.x))))

# 3. Pathway score = mean z-score of the genes in G_p, per patient
pathway_scores <- pathway_cols |>
  split(~pathway) |>
  map(\(x) rowMeans(z_mat[, x$col, drop = FALSE], na.rm = TRUE)) |>
  as_tibble() |>
  rename(
    score_hormone    = `Estrogen / hormone signaling`,
    score_pi3k_akt   = `PI3K-AKT-mTOR signaling`,
    score_cell_cycle = `Cell cycle / proliferation`,
    score_dna_repair = `DNA repair`
  )

df_full <- bind_cols(df, pathway_scores) |> 
  select(-c(score_dna_repair))

Clustering algorithm

I used k-mean clustering algorithm to cluster 2 subgroups based on 3 pathway score

Code 
# 1. Prepare the score matrix (complete cases only, rescaled so all 4 scores have equal weight)
score_df <- df_full |>
  mutate(patient_row = row_number()) |>
  select(patient_row, starts_with("score_")) |>
  drop_na()

score_mat <- score_df |>
  select(starts_with("score_")) |>
  scale()

# 2. Run k-means for k = 2 and k = 3
set.seed(123)
km2 <- kmeans(score_mat, centers = 2, nstart = 50)

# 3. Attach cluster labels back to the data
score_df <- score_df |>
  mutate(cluster_k2 = factor(km2$cluster))

df_clust <- df_full |>
  mutate(patient_row = row_number()) |>
  left_join(score_df |> select(patient_row, cluster_k2), by = "patient_row")

Binary variables

Code 
df_clust |>
  filter(!is.na(cluster_k2)) |>
  select(cluster_k2, er_status, her2_status, pr_status,
         chemotherapy, hormone_therapy, radio_therapy) |>
  mutate(across(
    c(chemotherapy, hormone_therapy, radio_therapy),
    ~ factor(.x, levels = c(0, 1), labels = c("No", "Yes"))
  )) |>
  mutate(across(-cluster_k2, as.character)) |>
  pivot_longer(-cluster_k2, names_to = "variable", values_to = "value") |>
  filter(!is.na(value)) |>
  count(cluster_k2, variable, value) |>
  group_by(cluster_k2, variable) |>
  mutate(pct = n / sum(n)) |>
  ungroup() |>
  filter(value %in% c("Yes", "Positive")) |>      # keep only the "yes/positive" category
  mutate(variable = recode(variable,
    er_status       = "ER positive",
    her2_status     = "HER2 positive",
    pr_status       = "PR positive",
    chemotherapy    = "Chemotherapy",
    hormone_therapy = "Hormone therapy",
    radio_therapy   = "Radiotherapy"
  )) |>
  ggplot(aes(x = cluster_k2, y = pct, fill = cluster_k2)) +
  geom_col(width = 0.6) +
  geom_text(aes(label = scales::percent(pct, accuracy = 1)), vjust = -0.3, size = 3) +
  facet_wrap(~variable, ncol = 3) +
  scale_y_continuous(labels = scales::percent, expand = expansion(mult = c(0, 0.12))) +
  labs(x = "Cluster", y = "Percentage within cluster") +
  theme_bw() +
  theme(legend.position = "none")
